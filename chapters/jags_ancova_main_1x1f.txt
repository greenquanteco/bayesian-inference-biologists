
    model{
        # priors
        ## intercept
        for(i in 1:nfac1){
            beta0[i] ~ dnorm(0, 0.001)
        }#i

        ## slope
        beta1 ~ dnorm(0, 0.001)

        ## residual SD and precision
        sigma ~ dlnorm(1, 0.25)
        tau <- 1 / (sigma^2)

        # likelihood
        for(i in 1:n){
            y[i] ~ dnorm(eta[i], tau)
            eta[i] <- beta0[fac1[i]] + beta1 * x1[i]
        }#i

    }#model
