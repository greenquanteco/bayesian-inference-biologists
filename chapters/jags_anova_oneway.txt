        # priors
        ## population means
        for(i in 1:ngroups){
            mu[i] ~ dnorm(0, 0.001)
            sigma[i] ~ dlnorm(1, 0.25)
            tau[i] <- 1/(sigma[i]^2)
        }#i

        # likelihood
        for(i in 1:n){
            y[i] ~ dnorm(mu[x[i]], tau[x[i]])
        }#i

    }#model
